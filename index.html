🚀 Hızlı Başlangıç
gettic.js'i yükle ve ilk botunu oluştur.

// Kurulum

npm install gettic.js

// Minimal Bot

const { Client } = require('gettic.js');

const bot = new Client({
    token: 'BOT_TOKEN',
    username: 'BenimBot',
    prefix: '/'
});

bot.on('ready', () => {
    console.log('Bot hazır!');
    bot.send('genel', 'Merhaba! 🤖');
});

bot.command('ping', ctx => ctx.reply(`🏓 ${bot.ping}ms`));
bot.connect();

// package.json

{
  "name": "benim-botum",
  "scripts": { "start": "node index.js" },
  "dependencies": { "gettic.js": "^1.0.0" }
}
🤖 Client
Client nesnesi tüm bot işlemlerinin merkezidir.

// Oluşturma

ParametreTipAçıklama
tokenstringPanelden alınan bot token'ı
usernamestringBot kullanıcı adı
prefixstringKomut öneki (varsayılan: /)

// Özellikler

AlanTipAçıklama
bot.usernamestringBot kullanıcı adı
bot.prefixstringKomut öneki
bot.readybooleanBağlantı kuruldu mu?
bot.pingnumberGecikme (ms)
bot.uptimenumberÇalışma süresi (ms)

// Metodlar

MetodAçıklama
bot.send(oda, mesaj)Odaya mesaj gönder
bot.joinRoom(oda)Odaya katıl
bot.leaveRoom(oda)Odadan ayrıl
bot.setName(isim)Bot adını değiştir
bot.setPrefix(önek)Prefix değiştir
bot.connect()Sunucuya bağlan
bot.destroy()Bağlantıyı kes
⌨ Komutlar
bot.command() ile komut tanımlama.

// Tekli Komut

bot.command('ping', (ctx) => {
    ctx.reply(`🏓 Pong! ${bot.ping}ms`);
});

// Toplu Komut

bot.commands([
    { name: 'ping', handler: ctx => ctx.reply('Pong!') },
    { name: 'sa',   handler: ctx => ctx.reply(`Selam ${ctx.sender}!`) }
]);

// Argümanlı Komut

bot.command('echo', (ctx) => {
    const text = ctx.args.join(' ');
    if (!text) return ctx.reply('❌ Bir şeyler yaz!');
    ctx.reply(text);
});

🔗 Webhook
Dış sistemlerden Gettic kanallarına mesaj gönder.

// WebhookClient

const { WebhookClient } = require('gettic.js');
const wh = new WebhookClient('WEBHOOK_URL');

await wh.send('Merhaba!');
await wh.sendEmbed({
    title: 'Başlık',
    description: 'Açıklama',
    fields: [{ name: 'Alan', value: 'Değer' }],
    footer: 'Alt yazı'
});

// HTTP POST (herhangi bir dil)

POST WEBHOOK_URL
Content-Type: application/json

{ "content": "Merhaba!" }

// Embed:
{ "embed": { "title": "...", "description": "...", "fields": [] } }

💡 Örnekler
Hazır bot örnekleri.

// Yardım Botu

bot.command('yardim', (ctx) => {
    ctx.reply(`📋 Komutlar:
/ping  — Gecikme testi
/sa    — Selamlama
/anket soru|a|b — Anket`);
});

// CI/CD Webhook

const wh = new WebhookClient(process.env.WEBHOOK_URL);

await wh.sendEmbed({
    title: '✅ Deploy Başarılı',
    description: `${process.env.BRANCH} → production`,
    fields: [
        { name: 'Commit', value: process.env.SHA.slice(0,7) },
        { name: 'Actor',  value: process.env.ACTOR }
    ]
});

// HTTP Panel

const http = require('http');

http.createServer((req, res) => {
    res.end(JSON.stringify({
        status: bot.ready ? 'online' : 'offline',
        ping: bot.ping + 'ms',
        uptime: Math.floor(bot.uptime / 1000) + 's'
    }));
}).listen(3000);

bot.connect();
