📚 Gettic.js Dökümantasyonu

Kurulum

npm install gettic.js

Bot Örneği

const { Client } = require('gettic.js');
const bot = new Client({ token: 'BOT_TOKEN' });
bot.on('ready', () => bot.send('genel', 'Merhaba!'));
bot.command('ping', ctx => ctx.reply('Pong!'));
bot.connect();

Webhook Örneği

const { WebhookClient } = require('gettic.js');
const webhook = new WebhookClient('WEBHOOK_URL');
webhook.send('Merhaba Gettic!');

REST API

const { RESTClient } = require('gettic.js');
const api = new RESTClient({ token: 'USER_TOKEN' });
const rooms = await api.getRooms();
const profile = await api.getProfile();

HTTP Panel

const http = require('http');
const server = http.createServer((req, res) => {
  if (req.url === '/ping') res.end('{"ping":"ok"}');
});
server.listen(3000);
