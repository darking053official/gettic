📖 Dökümantasyon

Basic Bot

Webhook
Olaylar
const { Client } = require('gettic.js');

const bot = new Client({
    token: 'BOT_TOKEN',
    username: 'BenimBot',
    prefix: '/'
});

bot.on('ready', () => {
    bot.send('genel', 'Merhaba Gettic! 🤖');
});

bot.connect();
bot.command('ping', (ctx) => {
    ctx.reply(`🏓 Pong! ${bot.ping}ms`);
});

bot.command('sa', (ctx) => {
    ctx.reply(`Selam ${ctx.sender} 👋`);
});

bot.command('top', (ctx) => {
    ctx.reply('🏆 Sıralama yapılıyor...');
});
const { WebhookClient } = require('gettic.js');

const wh = new WebhookClient('WEBHOOK_URL');
await wh.sendEmbed({
    title: 'Deploy Başarılı',
    description: 'Version 2.0 yayında!',
    fields: [{ name: 'Branch', value: 'main' }]
});
bot.on('message', (msg) => {
    console.log(`${msg.sender}: ${msg.content}`);
});

bot.on('user_join', (user) => {
    bot.send('genel', `${user} sunucuya katıldı!`);
});
