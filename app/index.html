const { Client } = require('gettic.js');\n\nconst bot = new Client({\n    token: 'BOT_TOKEN',\n    username: 'BenimBot',\n    prefix: '/'\n});\n\nbot.on('ready', () => {\n    console.log('Bot hazır!');\n    bot.send('genel', 'Merhaba Gettic!');\n});\n\nbot.command('ping', (ctx) => {\n    ctx.reply(`Pong! ${bot.ping}ms`);\n});\n\nbot.connect();
bot.command('merhaba', (ctx) => {\n    ctx.reply(`Merhaba ${ctx.sender}!`);\n});\n\nbot.command('top', (ctx) => {\n    const [page] = ctx.args;\n    ctx.reply(`Sıralama sayfası: ${page || 1}`);\n});\n\nbot.command('sil', async (ctx) => {\n    if (!ctx.member.roles.includes('admin')) {\n        return ctx.reply('Yetkiniz yok!');\n    }\n    const count = parseInt(ctx.args[0]) || 5;\n    await ctx.bulkDelete(count);\n    ctx.reply(`${count} mesaj silindi!`);\n});
const { WebhookClient } = require('gettic.js');\n\nconst webhook = new WebhookClient('WEBHOOK_URL');\n\nawait webhook.send('Deploy başarılı!');\n\nawait webhook.sendEmbed({\n    title: '**Yeni Güncelleme**',\n    description: 'v2.0.0 yayınlandı!',\n    color: '#6366f1',\n    fields: [\n        { name: 'Eklenenler', value: 'Bot paneli, webhook sistemi' },\n        { name: 'Düzeltmeler', value: 'Performans iyileştirmeleri' }\n    ],\n    footer: 'Gettic Team'\n});
bot.on('message', (message) => {\n    console.log(`${message.sender}: ${message.content}`);\n});\n\nbot.on('userJoin', (user) => {\n    bot.send('genel', `${user.username} sunucuya katıldı!`);\n});\n\nbot.on('voiceJoin', (user, channel) => {\n    console.log(`${user} ${channel} kanalına girdi`);\n});\n\nbot.on('messageDelete', (messageId) => {\n    console.log(`Mesaj silindi: ${messageId}`);\n});
bot.command('günlük', (ctx) => {\n    const amount = Math.floor(Math.random() * 500) + 100;\n    ctx.economy.add(ctx.senderId, amount);\n    ctx.reply(`Günlük ödülün: ${amount} coin!`);\n});\n\nbot.command('bakiye', (ctx) => {\n    const balance = ctx.economy.get(ctx.senderId);\n    ctx.reply(`Bakiyen: ${balance} coin`);\n});\n\nbot.command('transfer', (ctx) => {\n    const [target, amount] = ctx.args;\n    if (!target || !amount) return ctx.reply('Kullanım: /transfer @kullanici miktar');\n    ctx.economy.transfer(ctx.senderId, target, parseInt(amount));\n    ctx.reply(`${amount} coin gönderildi!`);\n});
