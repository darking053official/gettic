🚀 Hızlı Başlangıç

Gettic botunu birkaç dakikada çalıştır.

Kurulum

gettic.js NPM paketi ile botunu kolayca oluştur:

npm install gettic.js

Ya da bağımlılıkları manuel kur:

npm install socket.io-client axios

İlk Bot

const { Client } = require('gettic.js');

const bot = new Client({
    url: 'https://gettic-production.up.railway.app',
    token: 'BOT_TOKEN',   // Panelden al
    username: 'BenimBot',
    prefix: '/'
});

bot.on('ready', () => {
    console.log('✅ Bot hazır!');
    bot.send('genel', 'Merhaba! Ben bir Gettic botuyum 🤖');
});

bot.on('message', (msg) => {
    console.log(`${msg.senderName}: ${msg.content}`);
});

bot.command('ping', (ctx) => {
    ctx.reply(`🏓 Pong! Gecikme: ${bot.ping}ms`);
});

bot.connect();

package.json

{
  "name": "gettic-bot",
  "version": "1.0.0",
  "main": "index.js",
  "scripts": { "start": "node index.js" },

}

Başlatma

npm install

🔗 Webhook API

Dış sistemlerden Gettic kanallarına mesaj gönder.

WebhookClient Kullanımı

const { WebhookClient } = require('gettic.js');

const webhook = new WebhookClient(
    'https://gettic-production.up.railway.app/api/webhook/WEBHOOK_TOKEN'
);

// Basit mesaj
await webhook.send('Merhaba Gettic!');

// Embed mesaj
await webhook.sendEmbed({
    title: '🚀 Yeni Güncelleme!',
    description: 'v2.0 yayınlandı.',
    fields: [
        { name: 'Özellik', value: '✅ Eklendi' },
        { name: 'Hata', value: '🔧 Düzeltildi' }
    ],
    footer: 'Gettic Webhook'
});

HTTP ile Mesaj Gönderme

Herhangi bir dil/platform ile POST isteği atabilirsin:

POST https://gettic-production.up.railway.app/api/webhook/WEBHOOK_TOKEN

// Normal mesaj:

// Embed mesaj:
{
  "embed": {
    "title": "Başlık",
    "description": "Açıklama",
    "fields": [{ "name": "Alan", "value": "Değer" }]
  }
}

💡 Örnek Botlar

Hazır bot örnekleri.

Komut Botu

const { Client } = require('gettic.js');
const bot = new Client({ url: '...', token: 'TOKEN', username: 'KomutBot' });

bot.command('yardim', (ctx) => {
    ctx.reply(`📋 **Komutlar:**
/ping - Gecikme testi
/sa - Selam ver
/yardim - Bu menü
/anket soru|a|b - Anket başlat`);
});

bot.command('ping', (ctx) => ctx.reply(`🏓 Pong! ${bot.ping}ms`));
bot.command('sa', (ctx) => ctx.reply(`Aleyküm selam ${ctx.sender}! 👋`));

bot.command('anket', (ctx) => {
    const parts = ctx.args.join(' ').split('|');
    if (parts.length < 3) return ctx.reply('❌ /anket soru|seçenek1|seçenek2');
    ctx.reply(`📊 **${parts[0]}**\n🔵 ${parts[1]}\n🟢 ${parts[2]}`);
});

bot.connect();

HTTP Panel ile Bot

const { Client } = require('gettic.js');
const http = require('http');

const bot = new Client({ url: '...', token: 'TOKEN', username: 'PanelBot' });

http.createServer((req, res) => {
    res.setHeader('Content-Type', 'application/json');
    if (req.url === '/') {
        res.end(JSON.stringify({ status: bot.ready ? '🟢 Online' : '🔴 Offline', ping: bot.ping + 'ms' }));
    }
}).listen(3000);

bot.command('ping', ctx => ctx.reply('Pong!'));
bot.connect();
